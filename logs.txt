from datetime import datetime, timedelta, timezone
from typing import Optional, Dict, Any, Tuple

# Shared config
# ---------------------------------------------------------------------------

@dataclass(frozen=True)
class GateConfig:

    cooldown_min_minutes: int = 15
    cooldown_max_minutes: int = 30

# ---------------------------------------------------------------------------
# T3-B  Cooldown (15-30m per ticker/direction)
# ---------------------------------------------------------------------------

class CooldownGate:
    """
    Rejects a signal if the same (ticker, direction) fired inside the
    cooldown window. Window is itself adaptive in [15m, 30m] based on the
    volatility of the prior signal (higher ATR -> longer cooldown).
    """

    def __init__(self, cfg: GateConfig):
        self.cfg = cfg
        self._last: Dict[Tuple[str, int], Tuple[datetime, float]] = {}

    def _cooldown_for(self, prior_atr: float) -> timedelta:
        c = self.cfg
        # Normalise prior ATR into [0,1] using a 50-bps style soft ramp.
        # Low ATR -> min window; high ATR -> max window.
        span = max(c.cooldown_max_minutes - c.cooldown_min_minutes, 0)
        # 0.0 at atr<=0.003 (0.3%), 1.0 at atr>=0.02 (2%)
        ramp = min(max((prior_atr - 0.003) / (0.02 - 0.003), 0.0), 1.0)
        minutes = c.cooldown_min_minutes + ramp * span
        return timedelta(minutes=minutes)

    def check(self, sig: Signal) -> Tuple[bool, str]:
        key = (sig.ticker.upper(), sig.direction)
        prior = self._last.get(key)
        if prior is None:
            return True, "cooldown:first"

        prior_ts, prior_atr = prior
        window = self._cooldown_for(prior_atr)
        elapsed = sig.ts - prior_ts
        if elapsed < window:
            remaining = (window - elapsed).total_seconds() / 60.0
            return False, f"cooldown:blocked({remaining:.1f}m left)"
        return True, "cooldown:clear"

    def commit(self, sig: Signal) -> None:
        self._last[(sig.ticker.upper(), sig.direction)] = (sig.ts, sig.atr)


# ---------------------------------------------------------------------------
# T3-C  Bracket sanity validator
# ---------------------------------------------------------------------------

def bracket_sanity(sig: Signal, cfg: GateConfig) -> Tuple[bool, str]:
    """
    Reject if TP1 / SL are inverted, degenerate, or on the wrong side of
    entry for the signal direction, or if R:R is below cfg.min_rr.
    """
    e, tp, sl = sig.entry, sig.tp1, sig.sl

    if any(v is None for v in (e, tp, sl)):
        return False, "bracket:missing_price"

    if abs(tp - sl) < cfg.tick_epsilon or abs(tp - e) < cfg.tick_epsilon:
        return False, "bracket:degenerate"

    if sig.direction > 0:                       # long
        if not (tp > e > sl):
            return False, "bracket:inverted_long"
        risk, reward = e - sl, tp - e
    elif sig.direction < 0:                     # short
        if not (sl > e > tp):
            return False, "bracket:inverted_short"
        risk, reward = sl - e, e - tp
    else:
        return False, "bracket:bad_direction"

    if risk <= 0:
        return False, "bracket:nonpositive_risk"

    rr = reward / risk
    if rr < cfg.min_rr:
        return False, f"bracket:rr_too_low({rr:.2f}<{cfg.min_rr})"
    return True, f"bracket:ok(rr={rr:.2f})"


# ---------------------------------------------------------------------------
# T3-D  ATR stop scaling floor
# ---------------------------------------------------------------------------

def atr_stop_scaling(sig: Signal, cfg: GateConfig) -> Tuple[bool, str]:
    """
    Scale SL distance against ATR. Reject stops that are too tight
    (noise stop-out risk) or absurdly wide (bad R:R + slippage).
    """
    if sig.atr is None or sig.atr <= cfg.min_atr:
        return False, "atr:invalid"

    sl_dist = abs(sig.entry - sig.sl)
    ratio = sl_dist / sig.atr

    if ratio < cfg.atr_stop_floor_mult:
        return False, f"atr_stop:too_tight({ratio:.2f}<{cfg.atr_stop_floor_mult})"
    if ratio > cfg.atr_stop_ceiling_mult:
        return False, f"atr_stop:too_wide({ratio:.2f}>{cfg.atr_stop_ceiling_mult})"
    return True, f"atr_stop:ok({ratio:.2f}xATR)"


def scale_sl_to_atr(sig: Signal, cfg: GateConfig) -> float:
    """Push SL out to the floor, preserving direction. Returns new SL."""
    floor = cfg.atr_stop_floor_mult * sig.atr
    if sig.direction > 0:
        return min(sig.sl, sig.entry - floor)
    return max(sig.sl, sig.entry + floor)

# T3-E  EMA extension / distance filter
# ---------------------------------------------------------------------------

def ema_extension(sig: Signal, cfg: GateConfig) -> Tuple[bool, str]:
    """
    Reject entries that are already stretched far from the EMA stack
    (chasing). Also require EMA structure to agree with direction.
    """
    if sig.atr is None or sig.atr <= cfg.min_atr:
        return False, "ema_ext:no_atr"

    dist = abs(sig.price - sig.ema_slow)
    ext = dist / sig.atr

    if ext > cfg.ema_ext_hard_atr:
        return False, f"ema_ext:hard_reject({ext:.2f}>{cfg.ema_ext_hard_atr})"
    if ext > cfg.ema_ext_max_atr:
        return False, f"ema_ext:over_extended({ext:.2f}>{cfg.ema_ext_max_atr})"

    # Structure check
    if sig.direction > 0 and sig.ema_fast <= sig.ema_slow:
        return False, "ema_ext:bear_stack"
    if sig.direction < 0 and sig.ema_fast >= sig.ema_slow:
        return False, "ema_ext:bull_stack"

    return True, f"ema_ext:ok({ext:.2f}xATR)"

# T3-F  S/R buffer zone
# ---------------------------------------------------------------------------

def sr_buffer(sig: Signal, cfg: GateConfig) -> Tuple[bool, str]:
    """
    Reject if price sits inside the buffer band of an opposing S/R level.
    A level *ahead* of the target counts as a wall; being within the hard
    buffer is an unconditional reject.
    """
    if not sig.levels:
        return True, "sr:none"
    if sig.atr is None or sig.atr <= cfg.min_atr:
        return False, "sr:no_atr"

    soft = cfg.sr_buffer_atr * sig.atr
    hard = cfg.sr_hard_buffer_atr * sig.atr

    for lvl in sig.levels:
        d = abs(sig.price - lvl)
        if d <= hard:
            return False, f"sr:inside_hard_buffer({d:.4f}<={hard:.4f})"
        if d < soft:
            return False, f"sr:space_too_tight({d:.4f}<{soft:.4f})"

        # Blocked wall between entry and TP1
        between = min(sig.entry, sig.tp1) <= lvl <= max(sig.entry, sig.tp1)
        if between and d < 2.0 * soft:
            return False, f"sr:wall_before_tp1({lvl:.4f})"
    return True, "sr:clear"

# T3-G  GEX recalibration
# ---------------------------------------------------------------------------

def gex_recalibrated(sig: Signal, cfg: GateConfig) -> Tuple[bool, str]:
    """
    Recalibrated dealer-gamma gate.
      * walls only count if in the top (1-p) percentile of |gamma|
      * flip point must be respected as a soft regime boundary
      * direction must agree with the sign of net gamma near spot
    """
    g = sig.gex
    if not g:
        return True, "gex:absent"

    spot = g.get("spot", sig.price)
    net = g.get("net_gamma", 0.0)
    flip = g.get("flip", None)
    walls = g.get("walls") or []          # list of (strike, |gamma|)
    atr = sig.atr or cfg.min_atr

    # 1. Wall percentile filter
    if walls:
        mags = sorted((abs(m) for _, m in walls), reverse=True)
        k = max(int(round(cfg.gex_wall_percentile * (len(mags) - 1))), 0)
        thresh = mags[k]
        strong = [s for s, m in walls if abs(m) >= thresh and abs(m) >= cfg.gex_min_abs]
        for s in strong:
            # a wall directly in the path = reject
            in_path = min(sig.entry, sig.tp1) <= s <= max(sig.entry, sig.tp1)
            near = abs(spot - s) < 0.5 * atr
            if in_path and near:
                return False, f"gex:wall_block({s})"

    # 2. Flip tolerance
    if flip is not None:
        if abs(spot - flip) / max(spot, 1e-9) <= cfg.gex_flip_tolerance:
            return False, "gex:at_flip_noise"
        # Regime agreement
        if sig.direction > 0 and spot < flip:
            return False, "gex:below_flip_long"
        if sig.direction < 0 and spot > flip:
            return False, "gex:above_flip_short"

    # 3. Net gamma sign agreement
    if net != 0:
        if sig.direction > 0 and net < -cfg.gex_min_abs:
            return False, "gex:negative_gamma_long"
        if sig.direction < 0 and net > cfg.gex_min_abs:
            return False, "gex:positive_gamma_short"

    return True, "gex:ok"


# ---------------------------------------------------------------------------
# Orchestrator — the >80% gate
# ---------------------------------------------------------------------------

class MasterGate:
    """
    Runs the full stack. Cooldown is checked last (cheap + stateful)
    so it only consumes state on signals that already pass stateless gates.
    """

    def __init__(self, cfg: Optional[GateConfig] = None):
        self.cfg = cfg or GateConfig()
        self.cooldown = CooldownGate(self.cfg)

    def evaluate(self, sig: Signal) -> Dict[str, Any]:
        checks = [
            ("T3-C bracket", bracket_sanity(sig, self.cfg)),
            ("T3-D atr_stop", atr_stop_scaling(sig, self.cfg)),
            ("T3-E ema_ext", ema_extension(sig, self.cfg)),
            ("T3-F sr_buffer", sr_buffer(sig, self.cfg)),
            ("T3-G gex", gex_recalibrated(sig, self.cfg)),
            ("T3-B cooldown", self.cooldown.check(sig)),

        passed = not failures
        if passed:
            self.cooldown.commit(sig)
        return {
            "passed": passed,
            "failures": failures,
            "details": {name: reason for name, (_, reason) in checks},
        }
